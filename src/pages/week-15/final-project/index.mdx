## Project Brief

Build an integrated console deployment tracker that demonstrates the full Weeks 10-14 stack in one system. Your final project must show a custom class, controlled validation, <code>List&lt;T&gt;</code> collection management, and a readable controller flow that lets the instructor verify the system quickly.

class DeploymentRecord
{
    private string _title = string.Empty;
    private string _owner = string.Empty;
    private string _status = "Planned";

    public string Title
    {
        get => _title;
        set
        {
            if (string.IsNullOrWhiteSpace(value))
            {
                throw new ArgumentException("Title is required.");
            }

            _title = value.Trim();
        }
    }

    public string Owner
    {
        get => _owner;
        set
        {
            if (string.IsNullOrWhiteSpace(value))
            {
                throw new ArgumentException("Owner is required.");
            }

            _owner = value.Trim();
        }
    }

    public string Status
    {
        get => _status;
        set
        {
            string normalized = value.Trim();

            if (normalized != "Planned" &&
                normalized != "Ready" &&
                normalized != "Blocked" &&
                normalized != "Done")
            {
                throw new ArgumentException("Status must be Planned, Ready, Blocked, or Done.");
            }

            _status = normalized;
        }
    }

    public bool IsBlocked => Status == "Blocked";

    public DeploymentRecord(string title, string owner, string status)
    {
        Title = title;
        Owner = owner;
        Status = status;
    }

    public DeploymentRecord(string title, string owner)
        : this(title, owner, "Planned")
    {
    }
}` }
